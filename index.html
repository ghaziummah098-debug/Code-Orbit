function filterLanguages(){

  const value =
    document
      .getElementById("languageSearch")
      .value
      .toLowerCase()
      .trim();

  const filtered =
    Object.keys(languageData)
      .filter(lang =>
        lang.toLowerCase().includes(value)
      );

  renderLanguages(filtered);

}

  window.scrollTo(0, savedScrollPosition);

}

function loadLesson(index){

  currentLesson =
    Math.max(
      0,
      Math.min(
        index,
        lessonNames.length - 1
      )
    );

  const name =
    lessonNames[currentLesson];

  const data =
    languageData[currentLanguage];

  document
    .querySelectorAll(".lesson-item")
    .forEach(item =>
      item.classList.remove("active")
    );

  const active =
    document.getElementById(
      "lessonItem" + currentLesson
    );

  if(active){
    active.classList.add("active");
  }

  const percent =
    Math.round(
      ((currentLesson + 1) /
      lessonNames.length) * 100
    );

  document
    .getElementById("progressText")
    .textContent =
      `${currentLesson + 1} / ${lessonNames.length}`;

  document
    .getElementById("progressPercent")
    .textContent =
      percent + "%";

  document
    .getElementById("progressBar")
    .style.width =
      percent + "%";


  document
    .getElementById("lessonContent")
    .innerHTML =
      renderLesson(
        name,
        data,
        currentLesson
      );


  if(currentLesson <= 16){

    setEditorMode(
      currentLanguage
    );

  }

  updateLineNumbers();

}

    "Operators":
      "Operators perform calculations, comparisons and logical operations.",

      "Conditions let a program choose different actions based on rules.",

    "Loops":
      "Loops repeat code while a condition is true or for each item in a collection.",

    "Functions":
      "Functions group reusable logic into a named block.",

    "Arrays & Lists":

    "Classes":

    "Objects":
      "Objects represent data and behavior together.",

      "Inheritance allows one class to reuse or extend another class.",

    "Polymorphism":
      "Polymorphism allows the same interface or operation to behave differently for different objects.",

    "Encapsulation":
      "Encapsulation keeps internal implementation details controlled behind a public interface.",

    "File Handling":
